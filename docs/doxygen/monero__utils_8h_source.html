212 static std::vector<std::shared_ptr<monero_block>> get_blocks_from_txs(std::vector<std::shared_ptr<monero_tx_wallet>> txs) {
213 std::shared_ptr<monero_block> unconfirmed_block = nullptr; // placeholder for unconfirmed txs
214 std::vector<std::shared_ptr<monero_block>> blocks;
215 std::unordered_set<std::shared_ptr<monero_block>> seen_block_ptrs;
216 for (const std::shared_ptr<monero_tx_wallet>& tx : txs) {
217 if (tx->m_block == nullptr) {
218 if (unconfirmed_block == nullptr) unconfirmed_block = std::make_shared<monero_block>();
219 tx->m_block = unconfirmed_block;
220 unconfirmed_block->m_txs.push_back(tx);
221 }
222 std::unordered_set<std::shared_ptr<monero_block>>::const_iterator got = seen_block_ptrs.find(tx->m_block);
223 if (got == seen_block_ptrs.end()) {
224 seen_block_ptrs.insert(tx->m_block);
225 blocks.push_back(tx->m_block);
226 }
227 }
228 return blocks;
229 }

268 static void free(std::shared_ptr<monero_block> block) {
269 for (std::shared_ptr<monero_tx>& tx : block->m_txs) {
270 tx->m_block.reset();
271 monero_tx_wallet* tx_wallet = dynamic_cast<monero_tx_wallet*>(tx.get());
272 if (tx_wallet != nullptr) {
273 if (tx_wallet->m_tx_set != nullptr) tx_wallet->m_tx_set.reset();
274 if (tx_wallet->m_outgoing_transfer != nullptr) tx_wallet->m_outgoing_transfer->m_tx.reset();
275 for (std::shared_ptr<monero_transfer> transfer : tx_wallet->m_incoming_transfers) transfer->m_tx.reset();
276 for (std::shared_ptr<monero_output> output : tx_wallet->m_outputs) output->m_tx.reset();
277 for (std::shared_ptr<monero_output> input : tx_wallet->m_inputs) {
278 input->m_key_image.reset();
279 input->m_tx.reset();
280 }
281 }
282 monero_tx_query* tx_query = dynamic_cast<monero_tx_query*>(tx.get());
283 if (tx_query != nullptr) {
284 if (tx_query->m_transfer_query != nullptr) {
285 tx_query->m_transfer_query->m_tx_query.reset();
286 tx_query->m_transfer_query.reset();
287 }
288 if (tx_query->m_output_query != nullptr) {
289 tx_query->m_output_query->m_tx_query.reset();
290 tx_query->m_output_query.reset();
291 }
292 }
293 }
294 block.reset();
295 }
